}
bioc_install_if_missing <- function(pkgs) {

    install.packages("BiocManager", repos = "https://cloud.r-project.org")

    BiocManager::install(p, ask = FALSE, update = FALSE)
}

# expr-meta hizalama
sync_expr_meta <- function(expr, meta, stop_if_empty = TRUE, step = "unknown") {
  colnames(expr) <- as.character(colnames(expr))
  meta$barcode   <- as.character(meta$barcode)
  meta <- meta[!is.na(meta$barcode), , drop = FALSE]
  common <- intersect(colnames(expr), meta$barcode)
  if (length(common) == 0L) {
    m <- paste0("[SYNC] ortak barkod yok (", step, "). expr head: ",
                paste(utils::head(colnames(expr), 5), collapse=" | "),
                " | meta head: ",
                paste(utils::head(meta$barcode, 5), collapse=" | "))
    if (stop_if_empty) stop(m, call. = FALSE) else warning(m, call. = FALSE)
  }
  expr2 <- expr[, common, drop = FALSE]
  idx   <- match(common, meta$barcode)
  meta2 <- meta[idx, , drop = FALSE]
  dup <- duplicated(colnames(expr2))
  if (any(dup)) {
    expr2 <- expr2[, !dup, drop = FALSE]
    meta2 <- meta2[match(colnames(expr2), meta2$barcode), , drop = FALSE]
  }
  list(expr = expr2, meta = meta2)
}

get_expr_and_meta <- function(se) {
  if (inherits(se, "SummarizedExperiment")) {
    msg("SE akis.")
    col_anno <- as.data.frame(SummarizedExperiment::colData(se))
    if (!"barcode" %in% names(col_anno)) {
      if (!is.null(rownames(col_anno))) col_anno$barcode <- rownames(col_anno) else fail("colData barkod yok.")
    }
    col_anno$patient <- substr(col_anno$barcode, 1, 12)
    a_names <- SummarizedExperiment::assayNames(se)
    assay_use <- if ("read_count" %in% a_names) "read_count" else a_names[1]
    expr <- SummarizedExperiment::assay(se, assay_use)
    if ("shortLetterCode" %in% names(col_anno)) {
      code <- col_anno$shortLetterCode
      code[is.na(code) & grepl("Primary Tumor", col_anno$sample_type %||% "")] <- "TP"
      code[is.na(code) & grepl("Solid Tissue Normal", col_anno$sample_type %||% "")] <- "NT"
      col_anno$sample_type_code <- code
    } else {
      st <- col_anno$sample_type %||% ""
      col_anno$sample_type_code <- ifelse(grepl("Primary Tumor", st), "TP",
                                          ifelse(grepl("Solid Tissue Normal", st), "NT", NA))
    }
    return(list(expr = expr, meta = col_anno))
  }
  if (is.data.frame(se)) {
    msg("data.frame akis (miRNA tablo).")
    if (!"miRNA_ID" %in% names(se)) fail("'miRNA_ID' sutunu yok.")
    expr_df <- se
    rownames(expr_df) <- expr_df$miRNA_ID
    expr_df$miRNA_ID <- NULL
    read_cols <- grep("^read_count_", names(expr_df), value = TRUE)
    if (!length(read_cols)) {
      warn("read_count_ bulunamadi; sayisal sutunlar deneniyor.")
      read_cols <- names(expr_df)[sapply(expr_df, is.numeric)]
      if (!length(read_cols)) fail("Sayisal sutun yok.")
    }
    expr <- as.matrix(expr_df[, read_cols, drop = FALSE])
    barcodes <- sub("^read_count_", "", colnames(expr), perl = TRUE)
    barcodes[!grepl("^TCGA-", barcodes)] <- colnames(expr)[!grepl("^TCGA-", barcodes)]
    colnames(expr) <- barcodes
    patients <- substr(barcodes, 1, 12)
    scode <- stringr::str_extract(barcodes, "-(\\d\\d)[A-Z]")
    scode <- ifelse(is.na(scode), NA_character_, substr(scode, 2, 3))
    stype <- dplyr::case_when(
      scode == "01" ~ "TP",
      scode %in% c("11","21","22") ~ "NT",  # 21/22 normal varyantlari
      TRUE ~ NA_character_
    )
    meta <- tibble::tibble(barcode = barcodes, patient = patients, sample_type_code = stype)
    keep <- !is.na(meta$sample_type_code) & meta$sample_type_code %in% c("TP","NT")
    if (!any(keep)) fail("TP/NT ornek bulunamadi.")
    meta <- meta[keep, , drop = FALSE]
    expr <- expr[, meta$barcode, drop = FALSE]
    return(list(expr = expr, meta = meta))
  }
  fail("GDCprepare cikti sinifi taninamadi: ", paste(class(se), collapse=", "))
}

  get_keep_patients_by_subtype <- function() {
    sub <- tryCatch(TCGAquery_subtype("THCA"), error = function(e) NULL)
    keep_patients <- character(0)
    if (!is.null(sub)) {
      hist_cols <- names(sub)[grepl("hist|type", tolower(names(sub)))]
      if (length(hist_cols)) {
        wanted <- paste(wanted_histos, collapse="|")
        row_has <- apply(sub[, hist_cols, drop = FALSE], 1, function(r)
          any(stringr::str_detect(tolower(as.character(r)), wanted)))
        keep_patients <- unique(sub$patient[row_has])
      }
    }
    if (!length(keep_patients)) {
      clin <- tryCatch(GDCquery_clinic(project = project_id, type = "clinical"), error = function(e) NULL)
      if (!is.null(clin)) {
        idcol <- intersect(c("submitter_id","case_submitter_id","patient"), names(clin))
        if (length(idcol)) {
          clin$submitter_id <- clin[[idcol[1]]]
          txtcols <- names(clin)[sapply(clin, function(x) is.character(x) || is.factor(x))]
          txtcols <- txtcols[grepl("diagnos|hist|type|morph", tolower(txtcols))]
          if (length(txtcols)) {
            wanted <- paste(wanted_histos, collapse="|")
            any_hit <- apply(clin[, txtcols, drop = FALSE], 1, function(r)
              any(stringr::str_detect(tolower(paste(r, collapse=" ")), wanted)))
            keep_patients <- unique(clin$submitter_id[any_hit])
          }
        }
      }
    }
    keep_patients
  }
